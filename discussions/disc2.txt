void rotate(void *front, void *separator, void *end)
{
    char tmp[separator - front];
    memcpy(tmp, front, (separator - front) * sizeof(char));
    memmove(front, separator, (end - separator) * sizeof(char));
    memcpy(front + (end - separator), tmp, (separator - front) * sizeof(char));
}
